// frontend/js/dashboard.js const API_BASE_URL = "/api"; // Nginx proxies /api to backend async function analyzeResume() { const file = document.getElementById("resumeFile").files[0]; const jobDescription = document.getElementById("jobDescription").value; if (!file || !jobDescription) { alert("Please upload a resume and enter a job description."); return; } const formData = new FormData(); formData.append("file", file); formData.append("jobDescription", jobDescription); try { const response = await fetch(`${API_BASE_URL}/resume/analyze`, { method: "POST", headers: { Authorization: "Bearer " + localStorage.getItem("token") }, body: formData }); if (!response.ok) { throw new Error("Analysis failed"); } const result = await response.json(); // Render results document.getElementById("result").innerHTML = `

Candidate

Name: ${result.candidateName || "N/A"}

Email: ${result.candidateEmail || "N/A"}

Phone: ${result.candidatePhone || "N/A"}

Analysis

Match Score: ${result.matchScore}%

Matching Skills: ${result.matchingSkills?.join(", ") || "None"}

Missing Skills: ${result.missingSkills?.join(", ") || "None"}

Interview Questions

    ${result.interviewQuestions?.map(q => `
  • ${q}
    • `).join("") || "
  • No questions generated
    • "}
`; } catch (err) { console.error("Error:", err); alert("Something went wrong. Please try again."); } } // Attach event listener to button document.querySelector("button[onclick='analyzeResume()']").addEventListener("click", (e) => { e.preventDefault(); analyzeResume(); });
